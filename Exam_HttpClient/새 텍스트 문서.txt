async Task<string> GetWeatherByCityAsync(string city)
{
        
    try
    {
        //도시정보.
        string httpResult = await http.GetStringAsync($"https://geocoding-api.open-meteo.com/v1/search" +
                                                             $"?name={Uri.EscapeDataString(city)}&count=1&language=ko");
        var node = JsonNode.Parse(httpResult)!;
        Geo? geoResponse = node["results"]?[0].Deserialize<Geo>(jsonOptions); // 도시정보 

        if (geoResponse is null) { return "찾지 못함"; }        

        string httpResult2 = await http.GetStringAsync($"https://api.open-meteo.com/v1/forecast?latitude={Math.Round(geoResponse.latitude, 2)}" +
                                                              $"&longitude={Math.Round(geoResponse.longitude, 2)}&current=temperature_2m,wind_speed_10m" +
                                                              $"&daily=temperature_2m_max,temperature_2m_min&timezone=Asia%2FSeoul&forecast_days=3");
        var node2 = JsonNode.Parse(httpResult2)!;
        CurrentTemp? currentTemp = node2["current"]?.Deserialize<CurrentTemp>(jsonOptions);
        Daily? daily = node2["daily"]?.Deserialize<Daily>(jsonOptions);
        return $"{geoResponse.Name}    {currentTemp.Temperature}    {currentTemp.WindSpeed}    {daily.MinTemperature[1]} / {daily.MaxTemperature[1]}";
    }
    catch (HttpRequestException ex)
    {
        return ex.Message;
    }
}

async Task Step6HttpCall1(CancellationToken cts)
{
    //using var httpClient1 = new HttpClient();
    //JsonSerializerOptions jsonSerializerOptions = new() { PropertyNameCaseInsensitive = true };
    
    try
    {
        //도시정보.
        string httpResult = await http.GetStringAsync($"https://geocoding-api.open-meteo.comx/v1/search" +
                                                         $"?name=Seoul&count=1&language=ko",cts);
        var node = JsonNode.Parse(httpResult)!;
        Geo? geoResponse = node["results"]?[0].Deserialize<Geo>(jsonOptions); // 도시정보 

        if (geoResponse is null) Console.WriteLine("찾지 못함");

        //날씨정보.

        string httpResult2 = await http.GetStringAsync($"https://api.open-meteo.com/v1/forecast?latitude={Math.Round(geoResponse.latitude, 2)}" +
                                                              $"&longitude={Math.Round(geoResponse.longitude, 2)}&current=temperature_2m,wind_speed_10m" +
                                                              $"&daily=temperature_2m_max,temperature_2m_min&timezone=Asia%2FSeoul&forecast_days=3", cts);
        var node2 = JsonNode.Parse(httpResult2)!;
        CurrentTemp? currentTemp = node2["current"]?.Deserialize<CurrentTemp>(jsonOptions);
        Daily? daily = node2["daily"]?.Deserialize<Daily>(jsonOptions);
        Console.WriteLine($"{geoResponse.Name}    {currentTemp.Temperature}    {currentTemp.WindSpeed}    {daily.MinTemperature[1]} / {daily.MaxTemperature[1]}");
    }
    catch (HttpRequestException ex) when (ex.StatusCode is null) 
    {
        throw new HttpRequestException("연결 오류");
    }
    catch(HttpRequestException ex)
    {
        if ((int?)ex.StatusCode == 400)
        {
            throw new HttpRequestException("요청 요류");            
        }
        else
        {
            throw new HttpRequestException($"{ex.Message}");            
        }
    }
    catch(TaskCanceledException ex)
    {
        throw new HttpRequestException($"{ex.Message}");
    }
}



record Geo(string Name, double latitude, double longitude, string country);

record CurrentTemp(DateTime time, 
                   [property: JsonPropertyName("temperature_2m")] double Temperature,
                   [property: JsonPropertyName("wind_speed_10m")] double WindSpeed);

record Daily(List<DateTime> time,
             [property: JsonPropertyName("temperature_2m_max")] List<double> MaxTemperature,
             [property: JsonPropertyName("temperature_2m_min")] List<double> MinTemperature);
